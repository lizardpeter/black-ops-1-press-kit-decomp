cmake_minimum_required(VERSION 3.20)
project(t5_press_kit LANGUAGES CXX)
add_library(presskit STATIC src/launcher.cpp src/gallery.cpp src/navigation.cpp src/media.cpp src/timeline.cpp)
target_include_directories(presskit PUBLIC include)
target_compile_features(presskit PUBLIC cxx_std_17)
enable_testing()
add_executable(reconstruction_tests tests/reconstruction.cpp)
target_link_libraries(reconstruction_tests PRIVATE presskit)
add_test(NAME reconstruction COMMAND reconstruction_tests)

add_executable(navigation_tests tests/navigation.cpp)
target_link_libraries(navigation_tests PRIVATE presskit)
add_test(NAME navigation COMMAND navigation_tests)

add_executable(media_tests tests/media.cpp)
target_link_libraries(media_tests PRIVATE presskit)
add_test(NAME media COMMAND media_tests)

add_executable(timeline_tests tests/timeline.cpp)
target_link_libraries(timeline_tests PRIVATE presskit)
add_test(NAME timeline COMMAND timeline_tests)
